```tsx
post-card-list.tsx
"use client"

import React, { useState } from "react"
import { ArrowLeft, ArrowRight } from "lucide-react"
import { cn } from "@/lib/utils"

// ===== 类型定义 =====
interface Post {
  id: string
  title: string
  url: string
  date: string
  excerpt: string
  category?: string
}

interface PostCardListProps {
  posts: Post[]
  postsPerPage?: number
  className?: string
}

// ===== 单篇文章卡片 =====
function PostCard({ post }: { post: Post }) {
  return (
    <li className="group list-none mb-8 mx-8 transition-all duration-300 hover:pl-[30px] hover:border-l-[15px] hover:border-l-[#CE2323] border-l-[15px] border-l-transparent">
      <h2 className="text-[26px] leading-[48px] font-bold m-0">
        <a
          href={post.url}
          className="text-[#1122CC] no-underline hover:text-[#333366] hover:underline visited:text-[#1122CC]"
        >
          {post.title}
        </a>
      </h2>
      <span className="block my-2 text-[#8d8d8d] text-sm">{post.date}</span>
      <hr className="border-t border-[#eee] my-3" />
      <p className="text-[#555] leading-relaxed">{post.excerpt}</p>
      <p>
        <a
          href={post.url}
          className="text-[#1122CC] no-underline hover:text-[#333366] hover:underline visited:text-[#1122CC] font-medium"
        >
          阅读全文 &raquo;
        </a>
      </p>
      <hr className="border-t border-[#eee] my-3" />
    </li>
  )
}

// ===== 分页器 =====
function Paginator({
  currentPage,
  totalPages,
  onPageChange,
}: {
  currentPage: number
  totalPages: number
  onPageChange: (page: number) => void
}) {
  if (totalPages <= 1) return null

  return (
    <ul className="flex justify-center items-center gap-6 pt-4 border-t border-[#ddd] mt-4 list-none p-0">
      {currentPage > 1 && (
        <li>
          <button
            onClick={() => onPageChange(currentPage - 1)}
            className="flex items-center gap-1 text-[#1122CC] hover:text-[#333366] bg-transparent border-none cursor-pointer text-sm"
          >
            <ArrowLeft size={14} />
            更早
          </button>
        </li>
      )}
      {Array.from({ length: totalPages }, (_, i) => i + 1).map((page) => (
        <li key={page}>
          <button
            onClick={() => onPageChange(page)}
            className={cn(
              "bg-transparent border-none cursor-pointer text-sm px-2 py-1",
              page === currentPage
                ? "text-[#2E2435] font-bold"
                : "text-[#1122CC] hover:text-[#333366]"
            )}
          >
            {page}
          </button>
        </li>
      ))}
      {currentPage < totalPages && (
        <li>
          <button
            onClick={() => onPageChange(currentPage + 1)}
            className="flex items-center gap-1 text-[#1122CC] hover:text-[#333366] bg-transparent border-none cursor-pointer text-sm"
          >
            更新
            <ArrowRight size={14} />
          </button>
        </li>
      )}
    </ul>
  )
}

// ===== 主组件 =====
const PostCardList = React.forwardRef<HTMLDivElement, PostCardListProps>(
  ({ posts, postsPerPage = 8, className }, ref) => {
    const [currentPage, setCurrentPage] = useState(1)
    const totalPages = Math.ceil(posts.length / postsPerPage)
    const currentPosts = posts.slice(
      (currentPage - 1) * postsPerPage,
      currentPage * postsPerPage
    )

    return (
      <div ref={ref} className={cn("w-full", className)}>
        <ul className="p-0 m-0">
          {currentPosts.map((post) => (
            <PostCard key={post.id} post={post} />
          ))}
        </ul>
        <Paginator
          currentPage={currentPage}
          totalPages={totalPages}
          onPageChange={setCurrentPage}
        />
      </div>
    )
  }
)

PostCardList.displayName = "PostCardList"

export { PostCardList }
export type { Post, PostCardListProps }
```

```bash
lucide-react, clsx, tailwind-merge
```

实施指南
 1. 分析组件结构并确定所有必需的依赖项
 2. 审查组件的参数和状态
    - `posts`: Post[] — 文章数据数组，每项包含 id, title, url, date, excerpt, category
    - `postsPerPage`: number — 每页显示文章数，默认 8
    - `currentPage`: 内部状态，通过 useState 管理分页
 3. 确定是否需要上下文提供者（Context Providers）或钩子（Hooks），并进行安装
    - 不需要额外的 Context Provider
    - 使用了 React.useState 管理分页状态
 4. 需要确认的问题
  - 将向此组件传递哪些数据/属性？（posts 数组，postsPerPage 数量）
  - 是否有特定的状态管理需求？（仅内部分页状态，无需外部状态管理）
  - 是否需要任何资源文件（图片、图标等）？（使用 lucide-react 的 ArrowLeft/ArrowRight 图标）
  - 预期的响应式行为是什么？（Tailwind 响应式类，移动端全宽，桌面端 max-w-3xl 居中）
  - 悬停时左侧红色边框的颜色和宽度？（默认 #CE2323，15px，可通过修改 PostCard 中的 hover:border-l-[#CE2323] 调整）
  - 链接颜色方案？（默认蓝色 #1122CC，已访问同色，可通过修改 className 调整）
  - 在应用中使用此组件的最佳位置在哪里？（博客首页、文章列表页）

集成步骤
 0. 将上述所有代码复制粘贴到正确的目录中
 1. 安装外部依赖
 2. 准备文章数据：从 API、CMS 或本地 JSON 文件获取文章列表，转换为 Post[] 格式传入组件
 3. 如果需要自定义样式，修改 PostCard 中的 Tailwind 类名即可
 4. 如果需要按分类筛选，可在父组件中对 posts 数组进行 filter 后传入
