"use client"

import React, { useState } from "react"
import { ArrowLeft, ArrowRight } from "lucide-react"
import { cn } from "@/lib/utils"

// ===== 类型定义 =====
interface Post {
  id: string
  title: string
  url: string
  date: string
  excerpt: string
  category?: string
}

interface PostCardListProps {
  posts: Post[]
  postsPerPage?: number
  className?: string
}

// ===== 单篇文章卡片 =====
// 以下样式值全部提取自网站实际 css/index.css 和 index.html
function PostCard({ post }: { post: Post }) {
  return (
    <li
      className={cn(
        // 来源: ul.posts li — list-style:none; margin:0 2em 32px; transition:all .3s linear
        "list-none mx-[2em] mb-8",
        "transition-all duration-300 ease-linear",
        // 来源: ul.posts li:hover — padding-left:30px; border-left:15px solid #CE2323
        "border-l-[15px] border-l-transparent",
        "hover:pl-[30px] hover:border-l-[#CE2323]"
      )}
    >
      {/* 主标题 */}
      {/* 来源: h2 — font-size:26px; line-height:48px; font-weight:bold; margin:0 0 24px 0 */}
      {/* 来源: #index a:link — color:#12C */}
      {/* 来源: #index a:visited — color:#12C（已访问不变紫） */}
      {/* 来源: #index a:hover — color:#336; text-decoration:underline */}
      <h2 className="text-[26px] leading-[48px] font-bold m-0 mb-6">
        <a
          href={post.url}
          className="text-[#1122CC] no-underline visited:text-[#1122CC] hover:text-[#333366] hover:underline"
        >
          {post.title}
        </a>
      </h2>

      {/* 日期显示 */}
      {/* 来源: ul.posts .date — display:block; margin:8px 0; color:#8d8d8d */}
      <span className="block my-2 text-[#8d8d8d]">
        {post.date}
      </span>

      {/* 上分隔线 — 来源: index.html 中的 <hr> */}
      <hr className="my-3" />

      {/* 详情/摘要 */}
      {/* 来源: body — font-size:16px; color:#000000 */}
      {/* 来源: index.html — strip_html | strip_newlines | truncate: 160 */}
      <p className="text-base text-black leading-normal">
        {post.excerpt}
      </p>

      {/* 阅读全文链接 */}
      {/* 来源: #index a:link — color:#12C */}
      <p className="mt-2">
        <a
          href={post.url}
          className="text-[#1122CC] no-underline visited:text-[#1122CC] hover:text-[#333366] hover:underline"
        >
          阅读全文 &raquo;
        </a>
      </p>

      {/* 下分隔线 — 来源: index.html 中的 <hr> */}
      <hr className="my-3" />
    </li>
  )
}

// ===== 分页器 =====
// 来源: .paginator — border-top:1px solid #ddd; text-align:center; padding:15px 20px
// 来源: .paginator a — margin:0 12px
// 来源: .paginator .active — color:#2E2435; font-weight:bold
function Paginator({
  currentPage,
  totalPages,
  onPageChange,
}: {
  currentPage: number
  totalPages: number
  onPageChange: (page: number) => void
}) {
  if (totalPages <= 1) return null

  return (
    <ul className="border-t border-[#ddd] text-center py-[15px] px-5 list-none p-0 flex justify-center items-center gap-3">
      {currentPage > 1 && (
        <li>
          <button
            onClick={() => onPageChange(currentPage - 1)}
            className="mx-3 text-[#1122CC] hover:text-[#333366] bg-transparent border-none cursor-pointer inline-flex items-center gap-1"
          >
            <ArrowLeft size={14} />
            更早
          </button>
        </li>
      )}
      {Array.from({ length: totalPages }, (_, i) => i + 1).map((page) => (
        <li key={page}>
          <button
            onClick={() => onPageChange(page)}
            className={cn(
              "mx-3 bg-transparent border-none cursor-pointer",
              page === currentPage
                ? "text-[#2E2435] font-bold"
                : "text-[#1122CC] hover:text-[#333366]"
            )}
          >
            {page}
          </button>
        </li>
      ))}
      {currentPage < totalPages && (
        <li>
          <button
            onClick={() => onPageChange(currentPage + 1)}
            className="mx-3 text-[#1122CC] hover:text-[#333366] bg-transparent border-none cursor-pointer inline-flex items-center gap-1"
          >
            更新
            <ArrowRight size={14} />
          </button>
        </li>
      )}
    </ul>
  )
}

// ===== 主组件 =====
const PostCardList = React.forwardRef<HTMLDivElement, PostCardListProps>(
  ({ posts, postsPerPage = 8, className }, ref) => {
    const [currentPage, setCurrentPage] = useState(1)
    const totalPages = Math.ceil(posts.length / postsPerPage)
    const currentPosts = posts.slice(
      (currentPage - 1) * postsPerPage,
      currentPage * postsPerPage
    )

    return (
      <div ref={ref} className={cn("w-full", className)}>
        {/* 来源: ul.posts — 无额外padding/margin */}
        <ul className="p-0 m-0">
          {currentPosts.map((post) => (
            <PostCard key={post.id} post={post} />
          ))}
        </ul>
        <Paginator
          currentPage={currentPage}
          totalPages={totalPages}
          onPageChange={setCurrentPage}
        />
      </div>
    )
  }
)

PostCardList.displayName = "PostCardList"

export { PostCardList }
export type { Post, PostCardListProps }

实施指南
 1. 分析组件结构并确定所有必需的依赖项
 2. 审查组件的参数和状态
    - `posts`: Post[] — 文章数据数组，每项包含 id, title, url, date, excerpt, category
    - `postsPerPage`: number — 每页显示文章数，默认 8（来源: _config.yml paginate: 8）
    - `currentPage`: 内部状态，通过 useState 管理分页
 3. 确定是否需要上下文提供者（Context Providers）或钩子（Hooks），并进行安装
    - 不需要额外的 Context Provider
    - 使用了 React.useState 管理分页状态
 4. 需要确认的问题
  - 将向此组件传递哪些数据/属性？（posts 数组，postsPerPage 数量）
  - 是否有特定的状态管理需求？（仅内部分页状态，无需外部状态管理）
  - 是否需要任何资源文件（图片、图标等）？（使用 lucide-react 的 ArrowLeft/ArrowRight 图标）
  - 预期的响应式行为是什么？（Tailwind 响应式类，移动端全宽，桌面端 max-w-3xl 居中）
  - 在应用中使用此组件的最佳位置在哪里？（博客首页、文章列表页）
